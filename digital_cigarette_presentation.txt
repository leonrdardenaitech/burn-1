        const cigarette = document.getElementById('cigarette');

            atrophy = Math.min(100, Math.floor((hits / maxHits) * 100));

            atrophyFill.style.width = atrophy + '%';

            resetBtn.style.display = 'none';

        function createSparks() {
            const containerRect = container.getBoundingClientRect();
            const cigaretteRect = cigarette.getBoundingClientRect();
            
            // Spawn sparks at the right tip of the cigarette
            const x = cigaretteRect.right - containerRect.left;
            const y = cigaretteRect.top - containerRect.top + (cigaretteRect.height / 2);

            for (let i = 0; i < 8; i++) {
                const spark = document.createElement('div');
                spark.classList.add('spark-particle');
                spark.style.left = x + 'px';
                spark.style.top = y + 'px';
                container.appendChild(spark);

                const angle = (Math.random() - 0.5) * Math.PI;
                const speed = 2 + Math.random() * 3;
                let vx = Math.cos(angle) * speed;
                let vy = Math.sin(angle) * speed;
                let life = 1;

                const anim = setInterval(() => {
                    const cx = parseFloat(spark.style.left);
                    const cy = parseFloat(spark.style.top);
                    spark.style.left = (cx + vx) + 'px';
                    spark.style.top = (cy + vy) + 'px';
                    vy += 0.05; // gravity
                    life -= 0.05;
                    spark.style.opacity = life;

                    if (life <= 0) {
                        clearInterval(anim);
                        spark.remove();
                    }
                }, 30);
            }
        }
